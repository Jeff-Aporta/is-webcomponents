<is-maps>

Modo nativo (default): lienzo SVG con proyección equirectangular, sin dependencias externas. Modo engine="tile": iframe al proveedor de tiles (OpenStreetMap, etc.).

Colombia

Modo tile (OpenStreetMap embebido)

API

// Modo nativo (SVG)
const m = document.querySelector('is-maps');
// Atributos
// viewbox     "minLon,minLat,maxLon,maxLat"  viewport inicial
// interactive boolean — habilita pan/zoom con mouse/touch
//
// Marcadores
//   <is-map-marker lat="4.6" lon="-74.0" label="Bogotá"></is-map-marker>
//
// Modo tile
//   <is-maps engine="tile">
//     <script type="application/json">
//       { "tileUrl": "...", "zoom": 4, "bbox": "...", "center": "...",
//         "attribution": "..." }
//     </script>
//   </is-maps>
//
// Eventos
m.addEventListener('is-viewport',     e => console.log(e.detail));
m.addEventListener('is-marker-click', e => console.log(e.detail.marker));

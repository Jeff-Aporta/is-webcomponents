<is-data-grid>

Tabla con sort, selección, paginación y filtro de texto. Sin virtualización (v1).

Uso

const grid = document.querySelector('is-data-grid');
grid.columns = [
  { field: 'id', headerName: 'ID', width: 64 },
  { field: 'name', headerName: 'Nombre', sortable: true },
  { field: 'city', headerName: 'Ciudad' }
];
grid.rows = [
  { id: 1, name: 'Ana', city: 'Bogotá' },
  { id: 2, name: 'Luis', city: 'Medellín' }
];

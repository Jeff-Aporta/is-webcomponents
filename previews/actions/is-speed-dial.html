API

const dial = document.querySelector('is-speed-dial');
dial.open();         // abre
dial.close();        // cierra
dial.toggle();       // alterna
dial.isOpen;         // boolean
dial.addEventListener('is-select', e => console.log(e.detail.action.getAttribute('label')));
dial.addEventListener('is-toggle', e => console.log('open=', e.detail.open));

// Acciones: cada <is-speed-dial-action> emite is-select en el padre.
// Si tiene href, no se cierra el dial al elegir.

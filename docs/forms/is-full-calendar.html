API

const cal = document.querySelector('is-full-calendar');
cal.prev(); cal.next(); cal.today();
cal.setDate('2026-09-15');
cal.setView('week');
cal.addEventListener('is-day-click',  e => console.log(e.detail.date));
cal.addEventListener('is-event-click',e => console.log(e.detail.event));

// Atributos
// view          month | week | day
// date          ISO inicial (default hoy)
// first-day     0=Dom, 1=Lun (default 1)
// locale        (default 'es')
// hours-start, hours-end  rango en week/day (default 7..20)

// Datos
// { events: [ { id, title, date:'YYYY-MM-DD', start:'HH:MM', end?, color? } ] }

API


const r = document.querySelector('is-rte');
r.value;        // HTML actual
r.text;         // texto plano
r.focus();
r.exec('bold'); // bold, italic, underline, strike
r.format('h2'); // h1..h6, p, blockquote, pre
r.link();       // prompt → crea <a>
r.clear();      // sin formato, sin listas
r.undo(); r.redo();

r.addEventListener('is-input',  e => /* cada keystroke */);
r.addEventListener('is-change', e => console.log(e.detail.value));
r.addEventListener('is-blur',   () => /* perdió foco */);

// Atributos
// value, placeholder, autofocus, readonly
// toolbar   lista separada por comas

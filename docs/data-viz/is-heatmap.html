API

// Datos vía JSON o config
chart.config = {
  xLabels: [...],
  yLabels: [...],
  data: [[v, v, ...], ...]
};

// o por puntos
chart.config = { xLabels, yLabels, points: [{ x, y, v }, ...] };

chart.addEventListener('is-cell-hover', e => {
  console.log(`${e.detail.x} · ${e.detail.y} = ${e.detail.value}`);
});

// Atributos:
// x-label, y-label     títulos de eje
// color                brand | neutral | success | warning | danger | red-blue
// cell-radius          esquina (default 2)
// show-values          boolean, imprime el número dentro de cada celda
// legend-position      top | bottom | start | end | none
